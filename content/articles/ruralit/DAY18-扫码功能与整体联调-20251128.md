#### 1.3 扫码组件实现

```typescript
import { Html5Qrcode } from 'html5-qrcode';

interface QRScannerProps {
  onScanSuccess: (cardId: string) => void;
  onScanError?: (error: string) => void;
  onClose: () => void;
}

const QRScanner: React.FC<QRScannerProps> = ({
  onScanSuccess,
  onScanError,
  onClose
}) => {
  const scannerRef = useRef<Html5Qrcode | null>(null);
  
  useEffect(() => {
    const scanner = new Html5Qrcode('qr-reader');
    scannerRef.current = scanner;
    
    scanner.start(
      { facingMode: 'environment' },  // 后置摄像头
      {
        fps: 10,
        qrbox: { width: 250, height: 250 }
      },
      (decodedText) => {
        // 解析卡片ID
        const cardId = parseCardId(decodedText);
        if (cardId) {
          scanner.stop();
          onScanSuccess(cardId);
        }
      },
      (errorMessage) => {
        // 扫描中的错误（正常，不需要处理）
      }
    );
    
    return () => {
      scanner.stop().catch(() => {});
    };
  }, []);
  
  return (
    <div className="qr-scanner-overlay">
      <div className="scanner-header">
        <span>Point camera at card's QR code</span>
        <button onClick={onClose}>✕</button>
      </div>
      <div id="qr-reader" />
      <div className="scan-frame" />
    </div>
  );
};
```

#### 1.4 二维码格式定义

**卡片ID编码规则：**
```
格式：RURALIT_CARD_{cardNumber}_{uniqueId}
示例：RURALIT_CARD_001_A7B3C9
```

**解析函数：**
```typescript
function parseCardId(qrContent: string): string | null {
  const regex = /^RURALIT_CARD_(\d{3})_([A-Z0-9]{6})$/;
  const match = qrContent.match(regex);
  
  if (match) {
    return match[1]; // 返回卡片编号 "001"
  }
  return null;
}
```

#### 2.2 状态管理

```typescript
// 全局卡片状态
interface CardState {
  unlockedCards: string[];      // 已解锁卡片ID
  pendingUnlock: string | null; // 待解锁卡片（触发动画）
}

// Zustand store
const useCardStore = create<CardState>((set) => ({
  unlockedCards: [],
  pendingUnlock: null,
  
  unlockCard: (cardId: string) => {
    set((state) => ({
      unlockedCards: [...state.unlockedCards, cardId],
      pendingUnlock: cardId
    }));
  },
  
  clearPending: () => {
    set({ pendingUnlock: null });
  }
}));
```

#### 2.3 解锁动画触发

```typescript
// App.tsx
const { pendingUnlock, clearPending } = useCardStore();

useEffect(() => {
  if (pendingUnlock) {
    // 显示Rainbow Ready弹窗
    setShowUnlockModal(true);
  }
}, [pendingUnlock]);

// 弹窗关闭后清除pending状态
const handleModalClose = () => {
  setShowUnlockModal(false);
  clearPending();
};
```

## 今日关键洞察

### 1. html5-qrcode够用
- 不需要原生ML Kit
- 纯Web方案，跨平台一致
- 识别速度可接受（<1秒）

### 2. 状态管理很重要
- 扫码→解锁→动画→入库，涉及多个组件
- Zustand轻量且够用
- pendingUnlock模式解决动画触发问题

### 3. 资源释放是大坑
- 摄像头、MediaPipe都需要手动释放
- useEffect cleanup必须写
- 否则内存泄漏+摄像头占用
